cmake_minimum_required(VERSION 3.31)
project(SharedCanvasCoreConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6Qml TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6Quick TRUE)
find_package(iiSharedCanvas CONFIG REQUIRED COMPONENTS Core)
if(TARGET Qt6::Qml OR TARGET Qt6::Quick)
    message(FATAL_ERROR "Core imported a presentation dependency")
endif()
include(CTest)
add_executable(canvas_core_consumer main.cpp)
target_link_libraries(canvas_core_consumer PRIVATE iiSharedCanvas::Core)
add_test(NAME SharedCanvas.InstalledCore COMMAND canvas_core_consumer "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(canvas_publication_consumer ../PublicationMetadataTest.cpp)
target_link_libraries(canvas_publication_consumer PRIVATE iiSharedCanvas::Core iiFileProvider::Native ZLIB::ZLIB)
target_compile_definitions(canvas_publication_consumer PRIVATE IISHAREDCANVAS_TEST_OUTPUT_DIR="${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME SharedCanvas.InstalledPublicationMetadata COMMAND canvas_publication_consumer)
if(UNIX)
    if(APPLE)
        set(loader_path DYLD_LIBRARY_PATH)
    else()
        set(loader_path LD_LIBRARY_PATH)
    endif()
    set_tests_properties(SharedCanvas.InstalledCore SharedCanvas.InstalledPublicationMetadata PROPERTIES
        ENVIRONMENT_MODIFICATION "${loader_path}=path_list_prepend:$<TARGET_FILE_DIR:iiSharedCanvas::Core>;${loader_path}=path_list_prepend:$<TARGET_FILE_DIR:iiPaintEngine::iiPaintEngine>;${loader_path}=path_list_prepend:$<TARGET_FILE_DIR:iiFileProvider::iiFileProvider>;${loader_path}=path_list_prepend:$<TARGET_FILE_DIR:iiFileProvider::Native>")
endif()
