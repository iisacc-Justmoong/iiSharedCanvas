cmake_minimum_required(VERSION 3.31)
project(iiSharedCanvasInteractiveStrokeConsumer LANGUAGES CXX)
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
find_package(iiSharedCanvasNative 0.35.38 CONFIG REQUIRED)
find_package(iiFileProviderNative CONFIG REQUIRED)
if(NOT IISC_TEST_RESOURCES)
    message(FATAL_ERROR "IISC_TEST_RESOURCES must name the test resource directory")
endif()
add_executable(interactive-stroke-consumer ../InteractiveStrokeTest.cpp)
target_compile_features(interactive-stroke-consumer PRIVATE cxx_std_23)
target_compile_definitions(interactive-stroke-consumer PRIVATE IISHAREDCANVAS_TEST_OUTPUT_DIR="${IISC_TEST_RESOURCES}")
target_link_libraries(interactive-stroke-consumer PRIVATE iiSharedCanvas::Native iiFileProvider::Native)
enable_testing()
add_test(NAME iiSharedCanvas.InstalledInteractiveStroke COMMAND interactive-stroke-consumer)
set_tests_properties(iiSharedCanvas.InstalledInteractiveStroke PROPERTIES TIMEOUT 120)
