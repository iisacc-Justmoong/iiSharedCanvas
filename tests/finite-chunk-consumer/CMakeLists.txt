cmake_minimum_required(VERSION 3.24)
project(iiSharedCanvasFiniteChunkConsumer LANGUAGES CXX)
find_package(iiSharedCanvasNative 0.35.36 CONFIG REQUIRED)
if(NOT DEFINED IISC_FINITE_CONSUMER_RESOURCES)
  message(FATAL_ERROR "IISC_FINITE_CONSUMER_RESOURCES must point to an existing workspace resources directory")
endif()
add_executable(finite-chunk-consumer ../FiniteChunkRasterTest.cpp)
target_compile_features(finite-chunk-consumer PRIVATE cxx_std_23)
target_compile_definitions(finite-chunk-consumer PRIVATE IISHAREDCANVAS_TEST_OUTPUT_DIR="${IISC_FINITE_CONSUMER_RESOURCES}")
target_link_libraries(finite-chunk-consumer PRIVATE iiSharedCanvas::Native)
enable_testing()
add_test(NAME finite-chunk-consumer COMMAND finite-chunk-consumer)
